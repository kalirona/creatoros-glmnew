   241→function ChatTab({ onCreditsUpdate }: { onCreditsUpdate: (c: number) => void }) {
   242→  const [messages, setMessages] = useState<Array<{ role: 'user' | 'assistant'; content: string }>>([])
   243→  const [input, setInput] = useState('')
   244→  const [loading, setLoading] = useState(false)
   245→  const [tool, setTool] = useState('CHAT')
   246→  const scrollRef = useRef<HTMLDivElement>(null)
   247→
   248→  const TOOLS = [
   249→    { value: 'CHAT', label: 'General Assistant' },
   250→    { value: 'COURSE', label: 'Course Architect' },
   251→    { value: 'EMAIL', label: 'Email Copywriter' },
   252→    { value: 'SALES', label: 'Sales Page Writer' },
   253→    { value: 'BLOG', label: 'Blog Writer' },
   254→    { value: 'SOCIAL', label: 'Social Media' },
   255→    { value: 'SCRIPT', label: 'YouTube Script' },
   256→  ]
   257→
   258→  useEffect(() => {
   259→    if (scrollRef.current) scrollRef.current.scrollTop = scrollRef.current.scrollHeight
   260→  }, [messages])
   261→
   262→  const send = async () => {
   263→    const text = input.trim()
   264→    if (!text || loading) return
   265→    const newMessages = [...messages, { role: 'user' as const, content: text }]
   266→    setMessages(newMessages)
   267→    setInput('')
   268→    setLoading(true)
   269→    try {
   270→      const res = await fetch('/api/ai/chat', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ tool, messages: newMessages }) })
   271→      const raw = await res.text()
   272→      if (!res.ok) { let m = 'Failed'; try { const j = JSON.parse(raw); m = j.error } catch {} throw new Error(m) }
   273→      const data = JSON.parse(raw)
   274→      setMessages(prev => [...prev, { role: 'assistant', content: data.content || data.reply || '' }])
   275→      if (data.remainingCredits !== undefined) onCreditsUpdate(data.remainingCredits)
   276→      toast.success('AI responded', { description: `-${data.creditsUsed || 2} credits` })
   277→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Failed'); setMessages(prev => prev.slice(0, -1)) }
   278→    finally { setLoading(false) }
   279→  }
   280→
   281→  return (
   282→    <div className="flex flex-col h-[calc(100vh-300px)] min-h-[400px]">
   283→      <div className="flex items-center gap-2 mb-3">
   284→        <Select value={tool} onValueChange={setTool}>
   285→          <SelectTrigger className="w-52 h-9 text-sm"><SelectValue /></SelectTrigger>
   286→          <SelectContent>{TOOLS.map(t => <SelectItem key={t.value} value={t.value}>{t.label}</SelectItem>)}</SelectContent>
   287→        </Select>
   288→        {messages.length > 0 && <Button variant="ghost" size="sm" onClick={() => setMessages([])} className="text-sm"><RotateCcw className="h-3.5 w-3.5 mr-1.5" /> New Chat</Button>}
   289→      </div>
   290→
   291→      <div ref={scrollRef} className="flex-1 overflow-y-auto scroll-thin space-y-4 pr-2">
   292→        {messages.length === 0 ? (
   293→          <div className="flex flex-col items-center justify-center h-full text-center">
   294→            <div className="flex h-16 w-16 items-center justify-center rounded-2xl bg-primary/10 mb-4"><Sparkles className="h-8 w-8 text-primary" /></div>
   295→            <p className="text-base font-semibold">AI Chat</p>
   296→            <p className="text-sm text-muted-foreground mt-1 max-w-sm">Ask me anything about your business — course ideas, email copy, marketing strategy, and more.</p>
   297→            <div className="mt-5 flex flex-wrap gap-2 justify-center max-w-lg">
   298→              {['Give me 5 course ideas for creators', 'Write a welcome email for new students', 'Create a social media content plan', 'Generate a sales page outline'].map(ex => (
   299→                <button key={ex} onClick={() => setInput(ex)} className="rounded-full border px-3 py-1.5 text-xs text-muted-foreground hover:bg-muted transition">{ex}</button>
   300→              ))}
   301→            </div>
   302→          </div>
   303→        ) : (
   304→          messages.map((m, i) => (
   305→            <motion.div key={i} initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} className={cn('flex gap-3', m.role === 'user' && 'flex-row-reverse')}>
   306→              <Avatar className="h-9 w-9 shrink-0">
   307→                {m.role === 'assistant' ? <AvatarFallback className="bg-primary/15 text-primary"><Sparkles className="h-4 w-4" /></AvatarFallback>
   308→                  : <AvatarFallback className="bg-muted text-xs">You</AvatarFallback>}
   309→              </Avatar>
   310→              <div className={cn('flex-1 min-w-0 max-w-[80%] group', m.role === 'user' && 'flex justify-end')}>
   311→                <div className={cn('rounded-xl px-4 py-3 text-sm', m.role === 'user' ? 'bg-primary text-primary-foreground' : 'bg-muted')}>
   312→                  <div className="whitespace-pre-wrap">{m.content}</div>
   313→                </div>
   314→                {m.role === 'assistant' && (
   315→                  <div className="mt-1 flex items-center gap-1 opacity-0 group-hover:opacity-100 transition">
   316→                    <button onClick={() => { navigator.clipboard.writeText(m.content); toast.success('Copied') }} className="rounded p-1 text-xs text-muted-foreground hover:bg-muted" title="Copy"><Copy className="h-3 w-3" /></button>
   317→                  </div>
   318→                )}
   319→              </div>
   320→            </motion.div>
   321→          ))
   322→        )}
   323→        {loading && (
   324→          <div className="flex gap-3">
   325→            <Avatar className="h-9 w-9 shrink-0"><AvatarFallback className="bg-primary/15 text-primary"><Loader2 className="h-4 w-4 animate-spin" /></AvatarFallback></Avatar>
   326→            <div className="rounded-xl bg-muted px-4 py-3"><div className="flex gap-1"><span className="h-2 w-2 rounded-full bg-muted-foreground/50 animate-bounce" style={{ animationDelay: '0ms' }} /><span className="h-2 w-2 rounded-full bg-muted-foreground/50 animate-bounce" style={{ animationDelay: '150ms' }} /><span className="h-2 w-2 rounded-full bg-muted-foreground/50 animate-bounce" style={{ animationDelay: '300ms' }} /></div></div>
   327→          </div>
   328→        )}
   329→      </div>
   330→
   331→      <div className="mt-3 flex gap-2">
   332→        <Textarea value={input} onChange={(e) => setInput(e.target.value)} onKeyDown={(e) => { if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); send() } }} placeholder="Type your message... (Enter to send, Shift+Enter for new line)" rows={2} className="flex-1 resize-none text-sm" disabled={loading} />
   333→        <Button onClick={send} disabled={loading || !input.trim()} size="icon" className="h-auto px-3">{loading ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}</Button>
   334→      </div>
   335→    </div>
   336→  )
   337→}

   341→interface Tool { id: string; slug: string; name: string; description: string; icon: string; category: string; creditCost: number; outputType: string; isPro: boolean }
   342→const ICON_MAP: Record<string, React.ComponentType<{ className?: string }>> = { GraduationCap, FileText, Mail, ShoppingCart, LayoutTemplate, PenLine, Share2, Youtube, Package, MessageSquare, Sparkles }

   344→function DocumentsTab({ onCreditsUpdate, onNavigate }: { onCreditsUpdate: (c: number) => void; onNavigate: (module: string, subTab?: string) => void }) {
   345→  const [tools, setTools] = useState<Tool[]>([])
   346→  const [toolsLoading, setToolsLoading] = useState(true)
   347→  const [activeTool, setActiveTool] = useState<Tool | null>(null)
   348→  const [input, setInput] = useState('')
   349→  const [loading, setLoading] = useState(false)
   350→  const [result, setResult] = useState<any>(null)
   351→
   352→  useEffect(() => { fetch('/api/ai/generate').then(r => r.json()).then(d => { setTools(d.tools || []); setToolsLoading(false) }).catch(() => setToolsLoading(false)) }, [])
   353→
   354→  const generate = async () => {
   355→    if (!input.trim() || !activeTool || loading) return
   356→    setLoading(true); setResult(null)
   357→    try {
   358→      const controller = new AbortController(); const timeout = setTimeout(() => controller.abort(), 55000)
   359→      const res = await fetch('/api/ai/generate', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ toolSlug: activeTool.slug, input: input.trim() }), signal: controller.signal })
   360→      clearTimeout(timeout)
   361→      const raw = await res.text()
   362→      if (!res.ok) { let m = 'Failed'; try { const j = JSON.parse(raw); m = j.error } catch {} throw new Error(m) }
   363→      const data = JSON.parse(raw)
   364→      setResult(data)
