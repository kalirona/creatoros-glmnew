   101→export async function validateProviderKey(
   102→  providerSlug: ProviderSlug,
   103→  apiKey: string,
   104→  baseUrl?: string,
   105→): Promise<ValidationResult> {
   106→  if (!apiKey || apiKey.trim().length < 10) {
   107→    return { valid: false, message: 'API key is too short (minimum 10 characters).' }
   108→  }
   109→
   110→  // In production with real outbound access, we would:
   111→  //   1. GET {baseUrl}/models with the auth header
   112→  //   2. Parse the response and return real models
   113→  // In the sandbox we validate the key format and return the catalog.
   114→
   115→  // Basic format validation per provider
   116→  const formatCheck = checkKeyFormat(providerSlug, apiKey)
   117→  if (!formatCheck.ok) {
   118→    return { valid: false, message: formatCheck.message }
   119→  }
   120→
   121→  // Return the catalog for this provider
   122→  const models = MODEL_CATALOG[providerSlug] || []
   123→
   124→  return {
   125→    valid: true,
   126→    message: `Connected. ${models.length} models available.`,
   127→    models,
   128→    quotaRemaining: 'N/A (sandbox)',
   129→    providerVersion: getProviderVersion(providerSlug),
   130→  }
   131→}
   132→
   133→function checkKeyFormat(slug: ProviderSlug, key: string): { ok: boolean; message: string } {
   134→  const k = key.trim()
   135→  switch (slug) {
   136→    case 'openrouter':
   137→      return k.startsWith('sk-or-')
   138→        ? { ok: true, message: '' }
   139→        : { ok: false, message: 'OpenRouter keys start with "sk-or-"' }
   140→    case 'openai':
   141→      return k.startsWith('sk-')
   142→        ? { ok: true, message: '' }
   143→        : { ok: false, message: 'OpenAI keys start with "sk-"' }
   144→    case 'anthropic':
   145→      return k.startsWith('sk-ant-')
   146→        ? { ok: true, message: '' }
   147→        : { ok: false, message: 'Anthropic keys start with "sk-ant-"' }
   148→    case 'groq':
   149→      return k.startsWith('gsk_')
   150→        ? { ok: true, message: '' }
   151→        : { ok: false, message: 'Groq keys start with "gsk_"' }
   152→    case 'fal-ai':
   153→      return k.length >= 20
   154→        ? { ok: true, message: '' }
   155→        : { ok: false, message: 'Fal AI key is too short' }
   156→    case 'gemini':
   157→      return k.startsWith('AIza')
   158→        ? { ok: true, message: '' }
   159→        : { ok: false, message: 'Google AI keys start with "AIza"' }
   160→    case 'elevenlabs':
   161→      return k.length >= 20
   162→        ? { ok: true, message: '' }
   163→        : { ok: false, message: 'ElevenLabs key is too short' }
   164→    case 'deepgram':
   165→      return k.length >= 20
   166→        ? { ok: true, message: '' }
   167→        : { ok: false, message: 'Deepgram key is too short' }
   168→    case 'deepseek':
   169→      return k.startsWith('sk-')
   170→        ? { ok: true, message: '' }
   171→        : { ok: false, message: 'DeepSeek keys start with "sk-"' }
   172→    case 'together':
   173→      return k.length >= 20
   174→        ? { ok: true, message: '' }
   175→        : { ok: false, message: 'Together AI key is too short' }
   176→    case 'replicate':
   177→      return k.startsWith('r8_')
   178→        ? { ok: true, message: '' }
   179→        : { ok: false, message: 'Replicate tokens start with "r8_"' }
   180→    case 'glm':
   181→    case 'zai':
   182→      return k.length >= 10
   183→        ? { ok: true, message: '' }
   184→        : { ok: false, message: 'Key is too short' }
   185→    case 'custom':
   186→      return k.length >= 10
   187→        ? { ok: true, message: '' }
   188→        : { ok: false, message: 'Custom key is too short' }
   189→    default:
   190→      return { ok: true, message: '' }
   191→  }

   208→    zai: 'v4.6',
   209→    custom: 'compatible',
   210→  }
   211→  return versions[slug] || 'unknown'
   212→}

   214→// ─── Sync models to database ───────────────────────────────────────────────
   215→
   216→export async function syncProviderModels(providerId: string): Promise<SyncResult> {
   217→  const start = Date.now()
   218→  const provider = await db.aiProvider.findUnique({
   219→    where: { id: providerId },
   220→    include: { models: true },
   221→  })
   222→  if (!provider) {
   223→    return { status: 'failed', modelsFound: 0, modelsAdded: 0, modelsUpdated: 0, modelsRemoved: 0, modelsKept: 0, durationMs: 0, error: 'Provider not found' }
   224→  }
   225→
   226→  const slug = provider.slug as ProviderSlug
   227→  const discovered = MODEL_CATALOG[slug] || []
   228→
   229→  // Build a map of existing models by name
   230→  const existingMap = new Map(provider.models.map((m) => [m.name, m]))
   231→  const discoveredNames = new Set(discovered.map((d) => d.id))
   232→
   233→  let added = 0
   234→  let updated = 0
   235→  let removed = 0
   236→  let kept = 0
   237→
   238→  // Add or update discovered models
   239→  for (const d of discovered) {
   240→    const existing = existingMap.get(d.id)
   241→    if (!existing) {
   242→      // New model — insert
   243→      await db.aiModel.create({
   244→        data: {
   245→          providerId,
   246→          name: d.id,
   247→          displayName: d.name,
   248→          modality: d.modality,
   249→          contextWindow: d.contextWindow,
   250→          inputCostPer1k: d.inputCostPer1k,
   251→          outputCostPer1k: d.outputCostPer1k,
   252→          supportsVision: d.supportsVision,
   253→          supportsImage: d.supportsImage,
   254→          supportsAudio: d.supportsAudio,
   255→          supportsVideo: d.supportsVideo,
   256→          supportsEmbeddings: d.supportsEmbeddings,
   257→          supportsStreaming: d.supportsStreaming,
   258→          supportsJson: d.supportsJson,
   259→          supportsToolCalling: d.supportsToolCalling,
   260→          supportsReasoning: d.supportsReasoning,
   261→          providerTags: JSON.stringify(d.tags),
   262→          lastSyncedAt: new Date(),
   263→          isActive: true,
   264→        },
   265→      })
   266→      added++
   267→    } else {
   268→      // Existing model — update metadata but preserve custom pricing
   269→      if (existing.isCustomPricing) {
   270→        // Keep custom pricing, only update capability flags
   271→        await db.aiModel.update({
   272→          where: { id: existing.id },
   273→          data: {
   274→            displayName: d.name,
   275→            modality: d.modality,
   276→            contextWindow: d.contextWindow,
   277→            supportsVision: d.supportsVision,
   278→            supportsImage: d.supportsImage,
   279→            supportsAudio: d.supportsAudio,
   280→            supportsVideo: d.supportsVideo,
   281→            supportsEmbeddings: d.supportsEmbeddings,
   282→            supportsStreaming: d.supportsStreaming,
   283→            supportsJson: d.supportsJson,
   284→            supportsToolCalling: d.supportsToolCalling,
   285→            supportsReasoning: d.supportsReasoning,
   286→            providerTags: JSON.stringify(d.tags),
   287→            lastSyncedAt: new Date(),
   288→          },
   289→        })
   290→      } else {
   291→        // Update everything including pricing
   292→        await db.aiModel.update({
   293→          where: { id: existing.id },
   294→          data: {
   295→            displayName: d.name,
   296→            modality: d.modality,
   297→            contextWindow: d.contextWindow,
   298→            inputCostPer1k: d.inputCostPer1k,
   299→            outputCostPer1k: d.outputCostPer1k,
   300→            supportsVision: d.supportsVision,
   301→            supportsImage: d.supportsImage,
   302→            supportsAudio: d.supportsAudio,
   303→            supportsVideo: d.supportsVideo,
   304→            supportsEmbeddings: d.supportsEmbeddings,
   305→            supportsStreaming: d.supportsStreaming,
   306→            supportsJson: d.supportsJson,
   307→            supportsToolCalling: d.supportsToolCalling,
   308→            supportsReasoning: d.supportsReasoning,
   309→            providerTags: JSON.stringify(d.tags),
   310→            lastSyncedAt: new Date(),
   311→          },
   312→        })
   313→      }
   314→      updated++
   315→    }
   316→  }
   317→
   318→  // Remove models that are no longer in the catalog (but keep custom-priced ones)
   319→  for (const [name, existing] of existingMap) {
   320→    if (!discoveredNames.has(name) && !existing.isCustomPricing) {
   321→      await db.aiModel.delete({ where: { id: existing.id } })
   322→      removed++
   323→    } else if (!discoveredNames.has(name)) {
   324→      kept++
   325→    } else {
   326→      kept++
   327→    }
   328→  }
   329→
   330→  // Update provider sync metadata
   331→  await db.aiProvider.update({
   332→    where: { id: providerId },
   333→    data: { lastSyncAt: new Date() },
   334→  })
   335→
   336→  // Log sync history
   337→  await db.aiProviderSyncHistory.create({
   338→    data: {
   339→      providerId,
   340→      status: 'success',
   341→      modelsFound: discovered.length,
   342→      modelsAdded: added,
   343→      modelsUpdated: updated,
   344→      modelsRemoved: removed,
   345→      modelsKept: kept,
   346→      durationMs: Date.now() - start,
   347→    },
   348→  })
   349→
   350→  return {
   351→    status: 'success',
   352→    modelsFound: discovered.length,
   353→    modelsAdded: added,
   354→    modelsUpdated: updated,
   355→    modelsRemoved: removed,
   356→    modelsKept: kept,
   357→    durationMs: Date.now() - start,
   358→  }
   359→}
